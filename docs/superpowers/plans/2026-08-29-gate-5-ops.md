### Task 1: RequestContext en todas las Edge

**Files:**
- Create/Modify: `supabase/functions/_shared/request_context.ts`
- Modify: cada `supabase/functions/*/index.ts` (8 funciones)

```ts
export type RequestContext = { requestId: string };
export type EdgeLogFields = RequestContext & {
  tenantId: string | null
  userId: string | null
  fn: string
  outcome: 'ok' | 'error'
};

export function readRequestContext(req: Request): RequestContext {
  const header = req.headers.get("x-request-id");
  const requestId = header && header.length <= 64 ? header : crypto.randomUUID();
  return { requestId };
}

export function toEdgeLog(
  ctx: RequestContext,
  claims: { tenant_id?: string; sub?: string } | undefined,
  fn: string,
  outcome: "ok" | "error",
): EdgeLogFields {
  return {
    requestId: ctx.requestId,
    tenantId: claims?.tenant_id ?? null,
    userId: claims?.sub ?? null,
    fn,
    outcome,
  };
}

export function logEdge(
  fields: EdgeLogFields,
  extra?: Record<string, unknown>,
) {
  const { requestId, tenantId, userId, fn, outcome } = fields
  console.log(JSON.stringify({ request_id: requestId, tenant_id: tenantId, user_id: userId, fn, outcome, ...extra }))
}
```
